cmake_minimum_required(VERSION 3.20)
project(helo LANGUAGES C CXX)

# 构建产物只允许保存在仓库的独立构建批次目录中。
file(REAL_PATH "${CMAKE_SOURCE_DIR}/build" HELO_BUILD_ROOT)
file(REAL_PATH "${CMAKE_BINARY_DIR}" HELO_BINARY_ROOT)
string(FIND "${HELO_BINARY_ROOT}/" "${HELO_BUILD_ROOT}/" HELO_BUILD_PREFIX)
if(NOT HELO_BUILD_PREFIX EQUAL 0 OR HELO_BINARY_ROOT STREQUAL HELO_BUILD_ROOT)
    message(FATAL_ERROR "Use a unique build/<name>_<YYYYMMDD_HHMMSS> directory")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config IN LISTS CMAKE_CONFIGURATION_TYPES)
    string(TOUPPER "${config}" config_upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/bin/${config}")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/lib/${config}")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/lib/${config}")
endforeach()

add_library(helo_protocol INTERFACE)
target_include_directories(helo_protocol INTERFACE "${PROJECT_SOURCE_DIR}/hw/include")

include(CTest)
include(cmake/softfloat.cmake)
add_subdirectory(hw/model/func_model)
set(HELO_DUT "func" CACHE STRING "Verification DUT implementation")
set_property(CACHE HELO_DUT PROPERTY STRINGS func perf rtl)
if(BUILD_TESTING)
    add_subdirectory(hw/verif)
endif()
